# Выполнить в папке task-10

# Установить библиотеки один раз после развертывания

npm install


# Запустить сервер

npm start


# открыть страницу http://localhost:3000


# Вспомогательные команды

npm run seed          # заполнить базу демо-данными (если база пуста)
npm test              # автоматические unit-тесты
npm run e2e           # браузерные автотесты


# Требования

Node.js 23.5 или новее. Проверено на Node.js 26.
База данных — SQLite, отдельный сервер БД не нужен.
