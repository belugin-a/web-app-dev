# Выполнить в папке task-9

# Установить библиотеки один раз после развертывания

npm install

# Запустить сервер

npm start

# открыть страницу http://localhost:3000


# Вспомогательные команды

npm run seed          # заполнить базу демо-данными (если база пуста)
npm test              # автотесты


# Устройство проекта

server/
  app.js                  сборка Express-приложения
  server.js               точка входа
  config.js               порт и пути
  db.js                   подключение к SQLite, схема таблицы
  routes/api.js           СЛОЙ 1: маршрутизация
  services/               СЛОЙ 2: бизнес-логика
  repositories/           СЛОЙ 3: доступ к данным
  utils/errors.js         типизированные ошибки
public/                   одностраничный клиент
data/seed.js              демо-данные
data/app.db               файл базы (создаётся при первом запуске)


# Требования

Node.js 23.5 или новее: нужны встроенные модуль node:sqlite и метод
DatabaseSync.function() — без него не работает регистронезависимый поиск.
Проверено на Node.js 26.
База данных — SQLite, отдельный сервер не нужен.
